function loadContacts() {
var xhttp = new XMLHttpRequest();
xhttp.onreadystatechange = function() {
if (this.readyState == 4 && this.status == 200) {
var contacts = JSON.parse(this.responseText);
var table = document.getElementById("contactsBody");
for (var i = 0; i < contacts.length; i++) {
var row = table.insertRow(i);
var srno = row.insertCell(0);
var name = row.insertCell(1);
var residenceNumber = row.insertCell(2);
var mobileNumber = row.insertCell(3);
var address = row.insertCell(4);
srno.innerHTML = contacts[i].srno;
name.innerHTML = contacts[i].name;
residenceNumber.innerHTML = contacts[i].residenceNumber;
mobileNumber.innerHTML = contacts[i].mobileNumber;
address.innerHTML = contacts[i].address;
}
}
};
xhttp.open("GET", "readContacts.php", true);
xhttp.send();
}
