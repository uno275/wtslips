Q. 1) Create a HTML fileto insert text before and after a Paragraph using jQuery. [Hint : Use before( ) and after( )]

<!DOCTYPE html>
<html>
<head>
<title>Insert Text Before and After a Paragraph using jQuery</title>
<script src="https://code.jquery.com/jquery-3.6.0.min.js"></script>
<script>
$(document).ready(function(){
$("#btnBefore").click(function(){
$("p").before("<b>Before Text</b>");
});
$("#btnAfter").click(function(){
$("p").after("<i>After Text</i>");
});
});
</script>
</head>
<body>
<p>This is a paragraph.</p>
<button id="btnBefore">Insert Text Before</button>
<button id="btnAfter">Insert Text After</button>
</body>
</html>
