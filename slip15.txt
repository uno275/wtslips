$(document).ready(function() {
$("#searchBox").on("keyup", function() {
var searchText = $(this).val();
if (searchText != "") {
$.ajax({
url: "suggestions.php",
method: "POST",
data: {text: searchText},
dataType: "json",
success: function(response) {
var len = response.length;
$("#suggestions").empty();
for (var i = 0; i < len; i++) {
var suggestion = response[i];
$("#suggestions").append("<li>" + suggestion +
"</li>");
}
}
});
}
});
});
